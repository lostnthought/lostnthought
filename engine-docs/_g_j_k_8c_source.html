376static vec3 support_capsule(CapsuleData* cap, vec3 dir, mat4 inv) {
377 dir = mat4_mul_direction(inv, dir);
378 if (vec3_length(dir) <= 1e-6)
379 {
380 return cap->a;
381 }
382 float dot_a = vec3_dot(cap->a, dir);
383 float dot_b = vec3_dot(cap->b, dir);
384
385 vec3 furthest_line_point = (dot_a > dot_b) ? cap->a : cap->b;
386
387 vec3 normalized_dir = vec3_normalize(dir);
388 vec3 sphere_offset = vec3_scale(normalized_dir, cap->radius);
389
390 return vec3_add(furthest_line_point, sphere_offset);
391}

393static void support(vec3 * v, size_t * index, vec3 * verts, size_t cnt, mat4 inv, vec3 dir)
394{
395
396 float best_dot = -INFINITY;
397 size_t best_idx = 0;
398 vec3 best;
399 dir = mat4_mul_direction(inv, dir);
400
401 for(int i = 0; i < cnt; i++)
402 {
403 vec3 v = verts[i];
404 float d = vec3_dot(v, dir);
405
406 if(d > best_dot)
407 {
408 best_dot = d;
409 best_idx = i;
410 best = v;
411 }
412 }
413
414 *v = best;
415 *index = best_idx;

418SupportVertex support_diff(CollisionData a_cd, CollisionData b_cd, mat4 a_mat, mat4 b_mat, mat4 a_inv, mat4 b_inv, vec3 dir)
419{
420
421 SupportVertex p;
422 switch(a_cd.type)
423 {
424 case COL_TRIANGLE:
425 support(&p.a, &p.index_a, a_cd.data.hull.pts, a_cd.data.hull.count, a_inv, dir);
426 break;
427 case COL_HULL:
428 support(&p.a, &p.index_a, a_cd.data.hull.pts, a_cd.data.hull.count, a_inv, dir);
429 break;
430 case COL_CAPSULE:
431 p.a = support_capsule(&a_cd.data.capsule, dir, a_inv);
432 break;
433 }
434 switch(b_cd.type)
435 {
436 case COL_TRIANGLE:
437 support(&p.b, &p.index_b, b_cd.data.hull.pts, b_cd.data.hull.count, b_inv, vec3_scale(dir, -1));
438 break;
439 case COL_HULL:
440 support(&p.b, &p.index_b, b_cd.data.hull.pts, b_cd.data.hull.count, b_inv, vec3_scale(dir, -1));
441 break;
442 case COL_CAPSULE:
443 p.b = support_capsule(&b_cd.data.capsule, vec3_scale(dir, -1), b_inv);
444 break;
445 }
446 p.p = vec3_sub(mat4_mul_point(a_mat, p.a), mat4_mul_point(b_mat, p.b));
447 return p;
448}
