50static bool BVH_slab_test_aabb(RayHit * hit, AABB aabb, vec3 origin, vec3 dir, vec3 inv_dir)
51{
52
53 float tx1 = (aabb.min_x - origin.x) * inv_dir.x;
54 float tx2 = (aabb.max_x - origin.x) * inv_dir.x;
55 float ty1 = (aabb.min_y - origin.y) * inv_dir.y;
56 float ty2 = (aabb.max_y - origin.y) * inv_dir.y;
57 float tz1 = (aabb.min_z - origin.z) * inv_dir.z;
58 float tz2 = (aabb.max_z - origin.z) * inv_dir.z;
59 float txmin = fminf(tx1, tx2);
60 float txmax = fmaxf(tx1, tx2);
61 float tymin = fminf(ty1, ty2);
62 float tymax = fmaxf(ty1, ty2);
63 float tzmin = fminf(tz1, tz2);
64 float tzmax = fmaxf(tz1, tz2);
65
66 float t_enter = fmaxf(txmin, fmaxf(tymin, tzmin));
67
68 float t_exit = fminf(txmax, fminf(tymax, tzmax));
69
70 return (t_exit >= fmaxf(0.0f, t_enter) && t_enter < hit->t);

136bool ray_intersects_obb_with_point(OBB obb, vec3 origin, vec3 dir, vec3 * intersection_point) {
137
138 vec3 axis_x = (vec3){obb.rot.a0, obb.rot.a1, obb.rot.a2};
139 vec3 axis_y = (vec3){obb.rot.b0, obb.rot.b1, obb.rot.b2};
140 vec3 axis_z = (vec3){obb.rot.c0, obb.rot.c1, obb.rot.c2};
141
142 vec3 center = obb.center;
143
144 vec3 rel = vec3_sub(origin, center);
145
146 vec3 local_orig = {
147 vec3_dot(rel, axis_x),
148 vec3_dot(rel, axis_y),
149 vec3_dot(rel, axis_z)
150 };
151
152 vec3 norm_dir = dir;
153 norm_dir = vec3_normalize(norm_dir);
154
155 vec3 local_dir = {
156 vec3_dot(norm_dir, axis_x),
157 vec3_dot(norm_dir, axis_y),
158 vec3_dot(norm_dir, axis_z)
159 };
160 float tmin = -INFINITY;
161 float tmax = INFINITY;
162
163 if (fabsf(local_dir.x) < RAY_EPS) {
164 if (local_orig.x < -obb.extents.x || local_orig.x > obb.extents.x) return false;
165 } else {
166 float invd = 1.0f / local_dir.x;
167 float t1 = (-obb.extents.x - local_orig.x) * invd;
168 float t2 = ( obb.extents.x - local_orig.x) * invd;
169 if (t1 > t2) { float tmp = t1; t1 = t2; t2 = tmp; }
170 if (t1 > tmin) tmin = t1;
171 if (t2 < tmax) tmax = t2;
172 if (tmin > tmax) return false;
173 }
174
175 if (fabsf(local_dir.y) < RAY_EPS) {
176 if (local_orig.y < -obb.extents.y || local_orig.y > obb.extents.y) return false;
177 } else {
178 float invd = 1.0f / local_dir.y;
179 float t1 = (-obb.extents.y - local_orig.y) * invd;
180 float t2 = ( obb.extents.y - local_orig.y) * invd;
181 if (t1 > t2) { float tmp = t1; t1 = t2; t2 = tmp; }
182 if (t1 > tmin) tmin = t1;
183 if (t2 < tmax) tmax = t2;
184 if (tmin > tmax) return false;
185 }
186
187 if (fabsf(local_dir.z) < RAY_EPS) {
188 if (local_orig.z < -obb.extents.z || local_orig.z > obb.extents.z) return false;
189 } else {
190 float invd = 1.0f / local_dir.z;
191 float t1 = (-obb.extents.z - local_orig.z) * invd;
192 float t2 = ( obb.extents.z - local_orig.z) * invd;
193 if (t1 > t2) { float tmp = t1; t1 = t2; t2 = tmp; }
194 if (t1 > tmin) tmin = t1;
195 if (t2 < tmax) tmax = t2;
196 if (tmin > tmax) return false;
197 }
198
199 if (tmax < 0.0f) return false;
200
201 if (tmin < 0.0f)
202 {
203 *intersection_point = vec3_add(origin, vec3_scale(dir, tmax));
204 return true;
205 }
206
207 *intersection_point = vec3_add(origin, vec3_scale(dir, tmin));
208
209 return true;
210
211}

387static bool process_collision_node(Mesh * m, MeshNode * mn, BVHNode * node, vec3 proposed, CollisionPoints * p)
388{
389
390 MeshFace * fs = CAST_VEC(m->tris, MeshFace);
391 Point * v = CAST_VEC(m->vertices, Point);
392 size_t * li = m->bvh_indices;
393 int tris = 0;
394 for (int k = 0; k < node->count; k++)
395 {
396
397 MeshFace * f = &fs[li[k + node->first_index]];
398 mat4 model = mn->global_matrix;
399 mat4 tp = m4_transpose(model);
400 mat4 inv;
401 m4_inverse((float*)&inv, (float*)&tp);
402 vec3 A = mat4_mul_point(mn->global_matrix, v[f->indices[0]].pos);
403 vec3 B = mat4_mul_point(mn->global_matrix, v[f->indices[1]].pos);
404 vec3 C = mat4_mul_point(mn->global_matrix, v[f->indices[2]].pos);
405 vec3 nr = vec3_normalize(mat4_mul_direction(inv, f->normal));
406
407 tris++;
408 CollisionPoints pt;
409 if (collision(proposed, (vec3[]){A, B, C}, nr, &pt))
410 {
411 if (pt.d > p->d)
412 {
413 *p = pt;
414 p->bias_impulse = 0;
415 p->normal_impulse = 0;
416 p->tangent_impulse = 0;
417 p->d = pt.d;
418 p->normal = nr;
419 p->collision = true;
420 p->mnB = mn->id;
421 p->pcB = mn->pc;
422 }
423 }
424
425 }
426 return false;
427}
