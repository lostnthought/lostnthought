10float animation_length(Animation *anim) {

23void sample_animation(const AnimationSampler *sampler, float t, float *out_value) {
24 if (sampler->count == 0) return;
25
26 float duration = sampler->times[sampler->count - 1];
27 if (duration <= 0.0f) {
28 memcpy(out_value, sampler->values, sizeof(float) * sampler->elem_count);
29 return;
30 }
31 t = fmodf(t, duration);
32
33 int i = 0;
34 while (i < sampler->count - 1 && t >= sampler->times[i + 1]) {
35 i++;
36 }
37
38 if (i == sampler->count - 1) {
39 memcpy(out_value, &sampler->values[i * sampler->elem_count],
40 sizeof(float) * sampler->elem_count);
41 return;
42 }
43
44 float t0 = sampler->times[i];
45 float t1 = sampler->times[i + 1];
46 float alpha = (t - t0) / (t1 - t0);
47
48 const float *v0 = &sampler->values[i * sampler->elem_count];
49 const float *v1 = &sampler->values[(i + 1) * sampler->elem_count];
50
51 switch (sampler->interpolation) {
52 case STEP:
53 memcpy(out_value, v0, sizeof(float) * sampler->elem_count);
54 break;
55
56 case LINEAR:
57 for (int j = 0; j < sampler->elem_count; j++) {
58 out_value[j] = lerp(v0[j], v1[j], alpha);
59 }
60 break;
61
62 // not handled
63 case CUBICSPLINE:
64 memcpy(out_value, v0, sizeof(float) * sampler->elem_count);
65 break;
66 }
67}
