515static void heap_push(Vector * h) {
516 int i = h->count - 1;
517 OpenNode * buf = CAST_VEC(*h, OpenNode);
518
519 while (i > 0) {
520 int parent = (i - 1) / 2;
521 if (buf[parent].f <= buf[i].f) break;
522 heap_swap(&buf[parent], &buf[i]);
523 i = parent;
524 }
525}

527static OpenNode heap_pop(Vector *h) {
528
529 OpenNode * buf = CAST_VEC(*h, OpenNode);
530 OpenNode root = buf[0];
531 buf[0] = buf[--h->count];
532 int i = 0;
533 while (1) {
534 int left = 2*i + 1, right = 2*i + 2, smallest = i;
535 if (left < h->count && buf[left].f < buf[smallest].f) smallest = left;
536 if (right < h->count && buf[right].f < buf[smallest].f) smallest = right;
537 if (smallest == i) break;
538 heap_swap(&buf[i], &buf[smallest]);
539 i = smallest;
540 }
541 return root;
542}

578
579
580 WS->current_run++;
581 WS->path_count = 0;
582 init_vector(&WS->min_heap, 0, sizeof(OpenNode));

584 AudioPortal * portals = CAST_VEC(GLOBALS.audio_portals, AudioPortal);

587
588 AudioPortal * p1 = &portals[end];

591 start_node->i = start;
592 float f = vec3_length(vec3_sub(p0->pos, p1->pos));

595 WS->stack[start].f = f;
596 WS->stack[start].came_from = -1;
597 WS->stack[start].run_id = WS->current_run;
598 heap_push(&WS->min_heap);
599
600
601 while(WS->min_heap.count > 0)
602 {
603
604 OpenNode current = heap_pop(&WS->min_heap);
605 if (current.i == end)
606 {
607 reconstruct_path_aportal(WS, start, end);
608 return end;
609 }
610
611 AudioPortal * current_portal = &portals[current.i];
612 if (!current_portal->adjacent_portals.buffer) continue;

617 int neighbor = adj[i];

619 AudioPortal * neighbor_portal = &portals[neighbor];
620 float tenative_g = current.g + vec3_length(vec3_sub(current_portal->pos, neighbor_portal->pos));

622 bool did_search = (WS->stack[neighbor].run_id == WS->current_run);

628 OpenNode * neighbor_node = vector_expand(&WS->min_heap, 1);
629 neighbor_node->i = neighbor;

631 neighbor_node->f = f;

635 WS->stack[neighbor].f = f;
636 WS->stack[neighbor].g = tenative_g;
637 WS->stack[neighbor].run_id = WS->current_run;
638 heap_push(&WS->min_heap);

640 }

642 }

651 if (path_distance <= min_distance) return 1.0f;

1249static void init_bandpass(Biquad * filter, float sample_rate, float center_freq, float q)
1250{
1251
1252 memset(filter, 0, sizeof(Biquad));
1253 float w0 = 2.0f * M_PI * center_freq / sample_rate;
1254 float alpha = sinf(w0) / (2.0f * q);
1255
1256 float a0 = 1.0f + alpha;
1257 filter->a0 = a0;
1258 filter->b0 = alpha / a0;
1259 filter->b1 = 0.0f;
1260 filter->b2 = -alpha / a0;
1261 filter->a1 = -2.0f * cosf(w0) / a0;
1262 filter->a2 = (1.0f - alpha) / a0;
1263
1264}

1363static void extract_t60(float * edc_db, AudioProbe * probe, int band, float bin_duration_seconds, int last_sample) {
1364
1365 int start_idx = -1;
1366 int end_idx = -1;
1367
1368 for (int i = 0; i < last_sample; i++) {
1369 if (start_idx == -1 && edc_db[i] <= -5.0f) {
1370 start_idx = i;
1371 }
1372 if (end_idx == -1 && edc_db[i] <= -35.0f) {
1373 end_idx = i;
1374 break;
1375 }
1376 }
1377
1378 if (start_idx == -1 || end_idx == -1) {
1379 probe->rt60[band] = 0.0f;
1380 probe->slope[band] = 0.0f;
1381 return;
1382 }
1383
1384 float sum_x = 0.0f, sum_y = 0.0f, sum_xy = 0.0f, sum_xx = 0.0f;
1385 for (int i = start_idx; i <= end_idx; i++) {
1386 float x = i * bin_duration_seconds;
1387 float y = edc_db[i];
1388
1389 sum_x += x;
1390 sum_y += y;
1391 sum_xy += x * y;
1392 sum_xx += x * x;
1393 }
1394
1395 int n = end_idx - start_idx + 1;
1396 float denominator = (n * sum_xx) - (sum_x * sum_x);
1397 float slope = 0.0f;
1398 float t60 = 0.0f;
1399
1400 if (denominator != 0.0f) {
1401 slope = ((n * sum_xy) - (sum_x * sum_y)) / denominator;
1402
1403 t60 = -60.0f / slope;
1404 }
1405 probe->rt60[band] = t60;
1406 probe->slope[band] = slope;
1407
1408}
