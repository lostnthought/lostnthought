24#define DEGREES_TO_RAD (3.14159265358979323846 / 180.0)
25vec3 spherical_to_cartesian(float azimuth_deg, float elevation_deg, float radius) {
26
27 float azimuth_rad = azimuth_deg * DEGREES_TO_RAD;
28 float elevation_rad = elevation_deg * DEGREES_TO_RAD;
29
30 float projected_rad = radius * cosf(elevation_rad);
31
32 vec3 pos =
33 {
34 -projected_rad * sinf(azimuth_rad),
35 radius * sinf(elevation_rad),
36 -projected_rad * cosf(azimuth_rad)
37
38 };
39
40
41 return pos;
42}

197void rotate_sh(float * OUT, float SH[SH_MAX], mat3 rot)
198{
199
200 const static float s_c3 = 0.94617469575f;
201 const static float s_c4 = -0.31539156525f;
202 const static float s_c5 = 0.54627421529f;
203
204 const static float s_c_scale = 1.0 / 0.91529123286551084f;
205 const static float s_c_scale_inv = 0.91529123286551084f;
206
207 const static float s_rc2 = 1.5853309190550713f * s_c_scale;
208 const static float s_c4_div_c3 = s_c4 / s_c3;
209 const static float s_c4_div_c3_x2 = (s_c4 / s_c3) * 2.0;
210
211 const static float s_scale_dst2 = s_c3 * s_c_scale_inv;
212 const static float s_scale_dst4 = s_c5 * s_c_scale_inv;
213
214 float src[SH_MAX];
215 memcpy(src, SH, sizeof(float) * SH_MAX);
216
217 OUT[0] = src[0];
218 OUT[1] = rot.b1 * src[1] - rot.b2 * src[2] + rot.b0 * src[3];
219 OUT[2] = -rot.c1 * src[1] + rot.c2 * src[2] - rot.c0 * src[3];
220 OUT[3] = rot.a1 * src[1] - rot.a2 * src[2] + rot.a0 * src[3];
221
222
223
224 float sh0 = src[7] + src[8] + src[8] - src[5];
225 float sh1 = src[4] + s_rc2 * src[6] + src[7] + src[8];
226 float sh2 = src[4];
227 float sh3 = -src[7];
228 float sh4 = -src[5];
229
230 float r2x = rot.a0 + rot.a1;
231 float r2y = rot.b0 + rot.b1;
232 float r2z = rot.c0 + rot.c1;
233
234 float r3x = rot.a0 + rot.a2;
235 float r3y = rot.b0 + rot.b2;
236 float r3z = rot.c0 + rot.c2;
237
238 float r4x = rot.a1 + rot.a2;
239 float r4y = rot.b1 + rot.b2;
240 float r4z = rot.c1 + rot.c2;
241
242 float sh0_x = sh0 * rot.a0;
243 float sh0_y = sh0 * rot.b0;
244 float d0 = sh0_x * rot.b0;
245 float d1 = sh0_y * rot.c0;
246 float d2 = sh0 * (rot.c0 * rot.c0 + s_c4_div_c3);
247 float d3 = sh0_x * rot.c0;
248 float d4 = sh0_x * rot.a0 - sh0_y * rot.b0;
249
250 float sh1_x = sh1 * rot.a2;
251 float sh1_y = sh1 * rot.b2;
252 d0 += sh1_x * rot.b2;
253 d1 += sh1_y * rot.c2;
254 d2 += sh1 * (rot.c2 * rot.c2 + s_c4_div_c3);
255 d3 += sh1_x * rot.c2;
256 d4 += sh1_x * rot.a2 - sh1_y * rot.b2;
257
258 float sh2_x = sh2 * r2x;
259 float sh2_y = sh2 * r2y;
260 d0 += sh2_x * r2y;
261 d1 += sh2_y * r2z;
262 d2 += sh2 * (r2z * r2z + s_c4_div_c3_x2);
263 d3 += sh2_x * r2z;
264 d4 += sh2_x * r2x - sh2_y * r2y;
265
266 float sh3_x = sh3 * r3x;
267 float sh3_y = sh3 * r3y;
268 d0 += sh3_x * r3y;
269 d1 += sh3_y * r3z;
270 d2 += sh3 * (r3z * r3z + s_c4_div_c3_x2);
271 d3 += sh3_x * r3z;
272 d4 += sh3_x * r3x - sh3_y * r3y;
273
274 float sh4_x = sh4 * r4x;
275 float sh4_y = sh4 * r4y;
276 d0 += sh4_x * r4y;
277 d1 += sh4_y * r4z;
278 d2 += sh4 * (r4z * r4z + s_c4_div_c3_x2);
279 d3 += sh4_x * r4z;
280 d4 += sh4_x * r4x - sh4_y * r4y;
281
282 OUT[4] = d0;
283 OUT[5] = -d1;
284 OUT[6] = d2 * s_scale_dst2;
285 OUT[7] = -d3;
286 OUT[8] = d4 * s_scale_dst4;
287}

413static int SDLCALL hrtf_worker(void * params)
414{
415
416 HRTFThreadData * td = (HRTFThreadData*)params;
417
418 while(1)
419 {
420
421 if (td->stop_requested)
422 {
423 SDL_UnlockMutex(td->mutex);
424 break;
425 }
426
427 if ((!td->new_probe_ready && !td->stop_requested))
428 {
429 SDL_CondWait(td->work_cond, td->mutex);
430 continue;
431 }
432
433
434 PartitionTable * t = NULL;
435 SDL_AtomicLock(&td->ir_swap_lock);
436 {
437 PartitionTable * f = td->FRONT;
438 PartitionTable * a = CONVOLVER_NODE.ir;
439 PartitionTable * b = CONVOLVER_NODE.prev_ir;
440
441 for (int i = 0; i < PARTITION_BUFFERS; i++)
442 {
443 if (b != &td->tables[i] && f != &td->tables[i] && a != &td->tables[i])
444 {
445
446 t = &td->tables[i];
447 break;
448 }
449 }
450 }
451 SDL_AtomicUnlock(&td->ir_swap_lock);
452
453 if (t == NULL)
454 {
455 continue;
456 }
457
458 td->new_probe_ready = false;
459 update_player_ir(t);
460
461
462 SDL_AtomicLock(&td->ir_swap_lock);
463 {
464 td->FRONT = t;
465 }
466 SDL_AtomicUnlock(&td->ir_swap_lock);
467
468
469 }
470
471 return 1;
472
473}
