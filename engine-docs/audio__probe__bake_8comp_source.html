209void frisvad_basis(in vec3 n, out vec3 b1, out vec3 b2) {
210 if (n.z < -0.99998796f) {
211 b1.x = 0.0f; b1.y = -1.0f; b1.z = 0.0f;
212 b2.x = -1.0f; b2.y = 0.0f; b2.z = 0.0f;
213 return;
214 }
215
216 float a = 1.0f / (1.0f + n.z);
217 float b = -n.x * n.y * a;
218
219 b1.x = 1.0f - n.x * n.x * a;
220 b1.y = b;
221 b1.z = -n.x;
222
223 b2.x = b;
224 b2.y = 1.0f - n.y * n.y * a;
225 b2.z = -n.y;
226}

228vec3 get_bounce_direction(vec3 normal) {
229 float r1 = rand();
230 float r2 = rand();
231
232 float phi = 2.0 * 3.14159265 * r1;
233 float x = cos(phi) * sqrt(r2);
234 float y = sin(phi) * sqrt(r2);
235 float z = sqrt(1.0 - r2);
236 vec3 local_dir = vec3(x, y, z);
237
238 vec3 tangent;
239 vec3 bitangent;
240 if (abs(normal.z) > 0.999)
241 {
242
243 frisvad_basis(normal, tangent, bitangent);
244 }
245 else
246 {
247
248 vec3 up = abs(normal.z) < 0.999 ? vec3(0.0, 0.0, 1.0) : vec3(1.0, 0.0, 0.0);
249 tangent = normalize(cross(up, normal));
250 bitangent = cross(normal, tangent);
251 }
252
253 mat3 TBN = mat3(tangent, bitangent, normal);
254
255 return normalize(TBN * local_dir);

258vec3 generate_sphere_dir(int i, int N)
259{
260
261 float phi = 3.14159265 * (3.0f - sqrt(5.0f));
262
263 float y = 1.0f - 2.0f*(i+0.5f)/N;
264
265 float r = sqrt(1-y*y);
266
267 float theta = phi*i;
268
269 return vec3(cos(theta) * r, y, sin(theta) * r);

289RayHit ray_intersects_triangle(vec3 origin, vec3 dir, vec3 a, vec3 b, vec3 c)
290{
291
292 RayHit out_hit;
293
294 out_hit.hit = false;
295
296 vec3 edge1 = b-a;
297 vec3 edge2 = c-a;
298 vec3 h = cross(dir, edge2);
299 float dA = dot(edge1, h);
300
301 if (abs(dA) < 0.000001f) {
302 return out_hit;
303 }
304
305 float f = 1.0f / dA;
306 vec3 s = origin - a;
307 float u = f * dot(s, h);
308 if (u < 0.0f || u > 1.0f) {
309 return out_hit;
310 }
311
312 vec3 q = cross(s, edge1);
313 float v = f * dot(dir, q);
314 if (v < 0.0f || u + v > 1.0f) {
315 return out_hit;
316 }
317
318 float t = f * dot(edge2, q);
319 if (t > 0.0) {
320 out_hit.t = t;
321
322 float w = 1.0f - u - v;
323 out_hit.pos = a * w + b * u + c * v;
324 out_hit.barycentrics = vec3(w, u, v);
325 out_hit.hit = true;
326
327 return out_hit;
328 }
329
330 return out_hit;
331}

333bool slab_test(BVHNode n, vec3 pos, vec3 inv_dir) {
334 vec3 t1 = (n.min - pos) * inv_dir;
335 vec3 t2 = (n.max - pos) * inv_dir;
336
337 vec3 tmin = min(t1, t2);
338 vec3 tmax = max(t1, t2);
339
340 float t_near = max(max(tmin.x, tmin.y), tmin.z);
341 float t_far = min(min(tmax.x, tmax.y), tmax.z);
342 float closest_valid_t = min(hit.t, hit.blocked_t);
343 return (t_far >= max(0.0f, t_near) && t_near < closest_valid_t);
