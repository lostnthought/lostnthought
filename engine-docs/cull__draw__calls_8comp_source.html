99AABBBounds transform_aabb_by_model_matrix(MeshNode mn, mat4 mat)
100{
101
102 vec3 center = (mat * mn.center).xyz;
103
104 vec3 AX = (mat * vec4(1,0,0, 0.0)).xyz;
105 vec3 AY = (mat * vec4(0,1,0, 0.0)).xyz;
106 vec3 AZ = (mat * vec4(0,0,1, 0.0)).xyz;
107
108
109 vec3 new_extents;
110 new_extents.x = abs(AX.x) * mn.extents.x + abs(AY.x) * mn.extents.y + abs(AZ.x) * mn.extents.z;
111 new_extents.y = abs(AX.y) * mn.extents.x + abs(AY.y) * mn.extents.y + abs(AZ.y) * mn.extents.z;
112 new_extents.z = abs(AX.z) * mn.extents.x + abs(AY.z) * mn.extents.y + abs(AZ.z) * mn.extents.z;
113
114 AABBBounds aabb;
115
116 aabb.min.x = mn.center.x - new_extents.x;
117 aabb.min.y = mn.center.y - new_extents.y;
118 aabb.min.z = mn.center.z - new_extents.z;
119 aabb.max.x = mn.center.x + new_extents.x;
120 aabb.max.y = mn.center.y + new_extents.y;
121 aabb.max.z = mn.center.z + new_extents.z;

123 return aabb;
