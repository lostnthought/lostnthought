253vec2 parallax_mapping(sampler2D normal_map, vec2 texCoords, vec3 viewDir, out float final_depth)
254{
255
256 vec2 dx = dFdx(texCoords);
257 vec2 dy = dFdy(texCoords);
258 const float minLayers = 2.0;
259 const float maxLayers = 16.0;
260
261 float tx_width = float(textureSize(normal_map, 0).x);
262
263 vec2 dx2 = max(dx * dx * tx_width * tx_width, dy * dy * tx_width * tx_width);
264 float mip_level = 0.5 * log2(max(dx2.x, dx2.y));
265
266 float fade = clamp(1.0 - (mip_level / 3.0), 0.0, 1.0);
267
268
269 float current_max_layers = mix(minLayers, maxLayers, fade);
270
271
272 float numLayers = mix(current_max_layers, minLayers, abs(dot(vec3(0.0, 0.0, 1.0), viewDir)));
273
274 float layerDepth = 1.0 / numLayers;
275 float currentLayerDepth = 0.0;
276
277
278 float uv_scale = length(fwidth(texCoords));
279 vec2 P = viewDir.xy *height_scale;
280 vec2 deltaTexCoords = vec2(P.x, -P.y) / numLayers;
281
282 vec2 currentTexCoords = texCoords;
283 float freq = 10;
284 float currentDepthMapValue = textureGrad(normal_map, currentTexCoords, dx, dy).a;
285
286 int safety = 100;
287 int cur = 0;
288 while(currentLayerDepth < currentDepthMapValue && safety > cur)
289 {
290 currentTexCoords -= deltaTexCoords;
291
292
293 currentDepthMapValue = textureGrad(normal_map, currentTexCoords, dx, dy).a;
294 currentLayerDepth += layerDepth;
295 cur++;
296 }
297
298 vec2 prevTexCoords = currentTexCoords + deltaTexCoords;
299 float afterDepth = currentLayerDepth - currentDepthMapValue;
300 float current_depth = textureGrad(normal_map, currentTexCoords, dx, dy).a;
301 float prev_depth = textureGrad(normal_map, prevTexCoords, dx, dy).a;
302 float beforeDepth = (prev_depth) - currentLayerDepth + layerDepth;
303
304 float weight = afterDepth / (afterDepth + beforeDepth);
305 vec2 finalTexCoords = mix(currentTexCoords, prevTexCoords, weight);
306 final_depth = mix(current_depth, prev_depth, weight);
307
308 return finalTexCoords;
309}

311vec3 get_parallax_corrected_vector(vec3 worldPos, vec3 viewDir, vec3 normal, vec3 boxMin, vec3 boxMax, vec3 probeCenter) {
312

314
315 vec3 firstPlaneIntersect = (boxMax - worldPos) / reflectionDir;
316 vec3 secondPlaneIntersect = (boxMin - worldPos) / reflectionDir;

322 vec3 intersectPosition = worldPos + reflectionDir * distance;

326 return correctedReflDir;

354float distribution_GGX(float NdotH, float roughness)
355{
356 float a = roughness * roughness;
357 float a2 = a * a;
358 float d = (NdotH * a2 - NdotH) * NdotH + 1.0;
359 return a2 / (PI * d * d + 0.00001);
360}

369vec3 fresnel_schlick(float CosTheta, vec3 F0)
370{
371 float fc = 1.0 - CosTheta;
372 float fc2 = fc * fc;
373 float fc5 = fc2 * fc2 * fc;
374 return F0 + (1.0 - F0) * fc5;
375}

386float visibility(float NdotV, float NdotL, float roughness)
387{
388 float k = roughness * 0.5;
389 float gV = NdotV * (1.0 - k) + k;
390 float gL = NdotL * (1.0 - k) + k;
391 return 0.25 / (gV * gL + 0.00001);
392}

394vec3 calculate_light(vec3 N, vec3 V, vec3 L, vec3 albedo, float roughness, float metallic, vec3 light_color)
395{
396 vec3 H = normalize(V + L);
397
398 float NdotL = max(dot(N, L), 0.0);
399 float NdotV = max(dot(N, V), 0.0);
400 float NdotH = max(dot(N, H), 0.0);
401 float LdotH = max(dot(L, H), 0.0);
402
403 if (NdotL <= 0.0) return vec3(0.0);
404
405 vec3 F0 = mix(DIELECTRIC_F0, albedo, metallic);
406
407 float D = distribution_GGX(NdotH, roughness);
408 vec3 F = fresnel_schlick(LdotH, F0);
409 float vis = visibility(NdotV, NdotL, roughness);
410
411 vec3 specular = D * F * vis;
412
413 vec3 kS = F;
414 vec3 kD = vec3(1.0) - kS;
415 kD *= 1.0 - metallic;
416
417 vec3 diffuse = kD * albedo / PI;
418
419 return (diffuse + specular) * light_color * NdotL;
420}

424float normal_filtering(vec3 N, float roughness) {
425 vec3 dNdx = dFdx(N);
426 vec3 dNdy = dFdy(N);
427
428 float variance = dot(dNdx, dNdx) + dot(dNdy, dNdy);
429
430 float kernel_roughness = min(variance * 0.5, 0.5);
431
432 float alpha = roughness * roughness;
433 float filtered_alpha = clamp(alpha + kernel_roughness, 0.0, 1.0);
434
435 return sqrt(filtered_alpha);
436}

450float compute_probe_weight(vec3 world_pos, RProbe rp) {
451
452 vec3 dist_min = world_pos - rp.min;
453 vec3 dist_max = rp.max - world_pos;
454
455 vec3 min_dist = min(dist_min, dist_max);
456 float closest_wall_dist = min(min(min_dist.x, min_dist.y), min_dist.z);
457
458 if (closest_wall_dist <= 0.0f) {
459 return 0.0f;
460 }
461
462 vec3 fade_margin = vec3(5, 5, 5);
463
464 float margin = min(min(fade_margin.x, fade_margin.y), fade_margin.z);
465 float weight = closest_wall_dist / margin;
466
467 if (weight > 1.0f) weight = 1.0f;
468 if (weight < 0.0f) weight = 0.0f;
469
470
471 return weight;
472}

508float calculate_attenuation(float distance, float radius, float intensity) {
509 float attenuation = intensity / (distance + 1.0);
510
511 float factor = distance / (radius * radius);
512 float window = clamp(1.0 - factor * factor * factor * factor, 0.0, 1.0);
513 window = window * window;
514
515 return attenuation * window;
516}
