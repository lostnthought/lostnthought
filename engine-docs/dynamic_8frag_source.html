221vec2 parallax_mapping(sampler2D normal_map, vec2 texCoords, vec3 viewDir, out float final_depth)
222{
223
224 vec2 dx = dFdx(texCoords);
225 vec2 dy = dFdy(texCoords);
226 const float minLayers = 2.0;
227 const float maxLayers = 16.0;
228
229 float tx_width = float(textureSize(normal_map, 0).x);
230
231 vec2 dx2 = max(dx * dx * tx_width * tx_width, dy * dy * tx_width * tx_width);
232 float mip_level = 0.5 * log2(max(dx2.x, dx2.y));
233
234 float fade = clamp(1.0 - (mip_level / 3.0), 0.0, 1.0);
235
236
237 float current_max_layers = mix(minLayers, maxLayers, fade);
238
239
240 float numLayers = mix(current_max_layers, minLayers, abs(dot(vec3(0.0, 0.0, 1.0), viewDir)));
241
242 float layerDepth = 1.0 / numLayers;
243 float currentLayerDepth = 0.0;
244
245
246 float uv_scale = length(fwidth(texCoords));
247 vec2 P = viewDir.xy *height_scale;
248 vec2 deltaTexCoords = vec2(P.x, -P.y) / numLayers;
249
250 vec2 currentTexCoords = texCoords;
251 float freq = 10;
252 float currentDepthMapValue = textureGrad(normal_map, currentTexCoords, dx, dy).a;
253
254 int safety = 100;
255 int cur = 0;
256 while(currentLayerDepth < currentDepthMapValue && safety > cur)
257 {
258 currentTexCoords -= deltaTexCoords;
259
260
261 currentDepthMapValue = textureGrad(normal_map, currentTexCoords, dx, dy).a;
262 currentLayerDepth += layerDepth;
263 cur++;
264 }
265
266 vec2 prevTexCoords = currentTexCoords + deltaTexCoords;
267 float afterDepth = currentLayerDepth - currentDepthMapValue;
268 float current_depth = textureGrad(normal_map, currentTexCoords, dx, dy).a;
269 float prev_depth = textureGrad(normal_map, prevTexCoords, dx, dy).a;
270 float beforeDepth = (prev_depth) - currentLayerDepth + layerDepth;
271
272 float weight = afterDepth / (afterDepth + beforeDepth);
273 vec2 finalTexCoords = mix(currentTexCoords, prevTexCoords, weight);
274 final_depth = mix(current_depth, prev_depth, weight);
275
276 return finalTexCoords;
277}

279vec3 get_parallax_corrected_vector(vec3 worldPos, vec3 viewDir, vec3 normal, vec3 boxMin, vec3 boxMax, vec3 probeCenter) {

322float distribution_GGX(float NdotH, float roughness)
323{
324 float a = roughness * roughness;
325 float a2 = a * a;
326 float d = (NdotH * a2 - NdotH) * NdotH + 1.0;
327 return a2 / (PI * d * d + 0.00001);
328}

330vec3 fresnel_schlick(float CosTheta, vec3 F0)
331{
332 float fc = 1.0 - CosTheta;
333 float fc2 = fc * fc;
334 float fc5 = fc2 * fc2 * fc;
335 return F0 + (1.0 - F0) * fc5;
336}

347float visibility(float NdotV, float NdotL, float roughness)
348{
349 float k = roughness * 0.5;
350 float gV = NdotV * (1.0 - k) + k;
351 float gL = NdotL * (1.0 - k) + k;
352 return 0.25 / (gV * gL + 0.00001);
353}

355vec3 calculate_light(vec3 N, vec3 V, vec3 L, vec3 albedo, float roughness, float metallic, vec3 light_color)
356{
357 vec3 H = normalize(V + L);
358
359 float NdotL = max(dot(N, L), 0.0);
360 float NdotV = max(dot(N, V), 0.0);
361 float NdotH = max(dot(N, H), 0.0);
362 float LdotH = max(dot(L, H), 0.0);
363
364 if (NdotL <= 0.0) return vec3(0.0);
365
366 vec3 F0 = mix(DIELECTRIC_F0, albedo, metallic);
367
368 float D = distribution_GGX(NdotH, roughness);
369 vec3 F = fresnel_schlick(LdotH, F0);
370 float vis = visibility(NdotV, NdotL, roughness);
371
372 vec3 specular = D * F * vis;
373
374 vec3 kS = F;
375 vec3 kD = vec3(1.0) - kS;
376 kD *= 1.0 - metallic;
377
378 vec3 diffuse = kD * albedo / PI;
379
380 return (diffuse + specular) * light_color * NdotL;
381}

385float normal_filtering(vec3 N, float roughness) {
386 vec3 dNdx = dFdx(N);
387 vec3 dNdy = dFdy(N);
388
389 float variance = dot(dNdx, dNdx) + dot(dNdy, dNdy);
390
391 float kernel_roughness = min(variance * 0.5, 0.5);
392
393 float alpha = roughness * roughness;
394 float filtered_alpha = clamp(alpha + kernel_roughness, 0.0, 1.0);
395
396 return sqrt(filtered_alpha);
397}

411float compute_probe_weight(vec3 world_pos, RProbe rp) {
412
413 vec3 dist_min = world_pos - rp.min;
414 vec3 dist_max = rp.max - world_pos;
415
416 vec3 min_dist = min(dist_min, dist_max);
417 float closest_wall_dist = min(min(min_dist.x, min_dist.y), min_dist.z);
418
419 if (closest_wall_dist <= 0.0f) {
420 return 0.0f;
421 }
422
423 vec3 fade_margin = vec3(5, 5, 5);
424
425 float margin = min(min(fade_margin.x, fade_margin.y), fade_margin.z);
426 float weight = closest_wall_dist / margin;
427
428 if (weight > 1.0f) weight = 1.0f;
429 if (weight < 0.0f) weight = 0.0f;
430
431
432 return weight;
433}

469float calculate_attenuation(float distance, float radius, float intensity) {
470 float attenuation = intensity / (distance + 1.0);
471
472 float factor = distance / (radius * radius);
473 float window = clamp(1.0 - factor * factor * factor * factor, 0.0, 1.0);
474 window = window * window;
475
476 return attenuation * window;
477}
