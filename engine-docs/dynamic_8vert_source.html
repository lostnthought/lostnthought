174vec3 pos_to_pv_local(OBB obb, vec3 pos)
175{
176
177 vec3 local_pos = pos - obb.center;
178 mat3 rot = mat3(obb.axis_x, obb.axis_y, obb.axis_z);
179 vec3 unrotated = inverse(rot) * local_pos;
180 vec3 norm = unrotated / (obb.extents * 2.0f);
181 return norm + 0.5f;
182}
