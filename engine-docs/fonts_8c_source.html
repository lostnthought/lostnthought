74 int y = cJSON_GetObjectItem(ch, "y")->valueint;

77 g->xoffset = cJSON_GetObjectItem(ch, "xoffset")->valueint;
78 g->yoffset = cJSON_GetObjectItem(ch, "yoffset")->valueint;
79 g->xadvance = cJSON_GetObjectItem(ch, "xadvance")->valueint;
80
81 g->width = width;
82 g->height = height;
83 if (g->yoffset < highest_y && g->height > 0) highest_y = g->yoffset;
84 if (g->yoffset + g->height > font->highest_dist) font->highest_dist = g->yoffset + g->height;
85 g->u0 = (float)x / tex_w;
86 g->v0 = (float)y / tex_h;
87 g->u1 = (float)(x + width) / tex_w;
88 g->v1 = (float)(y + height) / tex_h;

92 cJSON_Delete(root);

130 float v0 = g->v0;
131 float u1 = g->u1;
132 float v1 = g->v1;

134 if (x > max_x) max_x = x;
135 if (y1 > max_y) max_y = y1;
136 if (x > tb->start.x + tb->w) {
137 y += f->highest_dist * scale;
138 x = tb->start.x;
139 }
140 }
