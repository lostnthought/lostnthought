33Font * read_json_msdf(const char * json_path, const char * png_path) {
34
35 FILE *f = fopen(json_path, "rb");
36 fseek(f, 0, SEEK_END);
37 long len = ftell(f);
38 rewind(f);
39
40 char *json_data = malloc(len + 1);
41 fread(json_data, 1, len, f);
42 json_data[len] = '\0';
43 fclose(f);
44
45 cJSON *root = cJSON_Parse(json_data);
46 free(json_data);
47 if (!root) {
48 fprintf(stderr, "Failed to parse font json\n");
49 return NULL;
50 }
51
52 cJSON *common = cJSON_GetObjectItem(root, "common");
53 int tex_w = cJSON_GetObjectItem(common, "scaleW")->valueint;
54 int tex_h = cJSON_GetObjectItem(common, "scaleH")->valueint;
55
56 Font * font = vector_expand(&FONTS, 1);
57 font->tex_w = tex_w;
58 font->tex_h = tex_h;
59 font->id = FONTS.count-1;
60 strncpy(font->name, png_path, 255);
61 font->name[255] = '\0';
62 cJSON *chars = cJSON_GetObjectItem(root, "chars");
63 int count = cJSON_GetArraySize(chars);
64 font->glyphs = malloc(sizeof(Glyph) * count);
65 int highest_y = INT_MAX;
66 int highest_dist = 1;
67
68 for (int i = 0; i < count; i++) {
69
70 cJSON * ch = cJSON_GetArrayItem(chars, i);
71 Glyph * g = &font->glyphs[i];
72
73 g->id = cJSON_GetObjectItem(ch, "id")->valueint;
74 int x = cJSON_GetObjectItem(ch, "x")->valueint;
75 int y = cJSON_GetObjectItem(ch, "y")->valueint;
76 int width = cJSON_GetObjectItem(ch, "width")->valueint;
77 int height = cJSON_GetObjectItem(ch, "height")->valueint;
78 g->xoffset = cJSON_GetObjectItem(ch, "xoffset")->valueint;
79 g->yoffset = cJSON_GetObjectItem(ch, "yoffset")->valueint;
80 g->xadvance = cJSON_GetObjectItem(ch, "xadvance")->valueint;
81
82 g->width = width;
83 g->height = height;
84 if (g->yoffset < highest_y && g->height > 0) highest_y = g->yoffset;
85 if (g->yoffset + g->height > font->highest_dist) font->highest_dist = g->yoffset + g->height;
86 g->u0 = (float)x / tex_w;
87 g->v0 = (float)y / tex_h;
88 g->u1 = (float)(x + width) / tex_w;
89 g->v1 = (float)(y + height) / tex_h;
90 }
91 font->highest_y = highest_y;
92 font->glyph_count = count;
93 cJSON_Delete(root);
94 return font;
95}

97void determine_text_box_dimensions(TextBox * tb, float * width, float * height){
98
99 int len = strlen(tb->text);
100
101 if (len <= 0) return;
102
103 float x = tb->start.x, y = tb->start.y;
104 Font * fonts = CAST_VEC(FONTS, Font);

106 int tex_w = f->tex_w;
107 int tex_h = f->tex_h;

110 float max_x = 0, max_y = 0;
111 Glyph * glyphs = f->glyphs;
112
113 for (int i = 0; i < len; i++){
114 char ch = tb->text[i];
115
116 Glyph *g = NULL;
117 for (int j = 0; j < f->glyph_count; j++) {
118 if (glyphs[j].id == ch) {
119 g = &glyphs[j];
120 break;
121 }
122 }
123 if (!g) continue;
124
125 float x0 = x + g->xoffset * scale;
126 float y0 = (y - 7 * scale) + g->yoffset * scale;
127 float x1 = x0 + g->width * scale;
128 float y1 = y0 + g->height * scale;
129
130 float u0 = g->u0;
131 float v0 = g->v0;
132 float u1 = g->u1;
133 float v1 = g->v1;
134 x += g->xadvance * scale;
135 if (x > max_x) max_x = x;
136 if (y1 > max_y) max_y = y1;
137 if (x > tb->start.x + tb->w) {
138 y += f->highest_dist * scale;
139 x = tb->start.x;
140 }
141 }
142 if (width) *width = max_x - tb->start.x;
143 if (height) *height = max_y - tb->start.y;
144
145 float dx = (max_x - tb->start.x) / 2 + tb->start.x;
146 float dy = fabsf(max_y - tb->start.y) / 2 + tb->start.y;
