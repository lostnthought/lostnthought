124void find_least_parallel_axis(vec3 normal, vec3 * u_dir, vec3 * v_dir)
125{
126 vec3 n = vec3_normalize(normal);
127
128 float dx = fabsf(vec3_dot(n, (vec3){1,0,0}));
129 float dy = fabsf(vec3_dot(n, (vec3){0,1,0}));
130 float dz = fabsf(vec3_dot(n, (vec3){0,0,1}));
131
132 vec3 helper;
133
134 if(dx < dy && dx < dz)
135 helper = (vec3){1,0,0};
136 else if(dy < dz)
137 helper = (vec3){0,1,0};
138 else
139 helper = (vec3){0,0,1};
140 vec3 up = (vec3){0,0,1};
141 if (fabsf(n.z) >= 0.999f) up = (vec3){1,0,0};
142
143 *u_dir = vec3_normalize(vec3_cross(up, n));
144 *v_dir = vec3_cross(n, *u_dir);
145}

147LightmapRect allocate_lightmap_rect(LightmapAtlas *atlas, int w_px, int h_px) {
148
149 int padding = 2;
150
151
152 int total_w = w_px + 2 * padding;
153 int total_h = h_px + 2 * padding;
154 int blocks_w = (total_w + atlas->block_size + 1) / atlas->block_size;

156 int blocks_x = atlas->width / atlas->block_size;
157 int blocks_y = atlas->height / atlas->block_size;

159 for (int y = 0; y <= blocks_y - blocks_h; y++) {
160 for (int x = 0; x <= blocks_x - blocks_w; x++) {
161 bool fits = true;
162 for (int dy = 0; dy < blocks_h && fits; dy++) {
163 for (int dx = 0; dx < blocks_w; dx++) {
164 int idx = (y + dy) * blocks_x + (x + dx);
165 if (atlas->occupied[idx]) {
166 fits = false;
167 break;
168 }
169 }
170 }
171
172 if (fits) {
173 for (int dy = 0; dy < blocks_h; dy++) {
174 for (int dx = 0; dx < blocks_w; dx++) {
175 int idx = (y + dy) * blocks_x + (x + dx);
176 atlas->occupied[idx] = 1;
177 }
178 }
179
180 return (LightmapRect){
181 .u = x * atlas->block_size + padding,
182 .v = y * atlas->block_size + padding,
183 .w = w_px + 2 * padding,
184 .h = h_px + 2 * padding,
185 .success = 1,

298 }
299 }

306 int texel_width = MAX(ceilf(range_u * EDITOR_SETTINGS.texels_per_uv_unit * (EDITOR_SETTINGS.mfactor_enabled ? mfactor : 1)), 6);

317
318 int index = 0;
319
320 for (int i = 0; i < uvi->tris.count; i++)
321 {
322
323 MeshFace * mf = &mfs[tris[i]];
324 mn->uv_map[tris[i]].uv_idx = i;
325
326
327
328 for (int j = 0; j < 3; j++) {
329
330 float pu = uv_pairs[uv_pair_indices[index]].pu;
331 float pv = uv_pairs[uv_pair_indices[index]].pv;
332

334 float norm_v = (pv - min_v) / (range_v > 1e-6f ? range_v : 1.0f);
