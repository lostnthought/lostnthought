403bool ray_intersects_triangle(vec3 origin, vec3 dir, vec3 a, vec3 b, vec3 c, out RayHit out_hit)
404{
405
406 out_hit.hit = false;
407
408 vec3 edge1 = b-a;
409 vec3 edge2 = c-a;
410 vec3 h = cross(dir, edge2);
411 float dA = dot(edge1, h);
412
413 if (abs(dA) < 0.000001f) {
414 return false;
415 }
416
417 float f = 1.0f / dA;
418 vec3 s = origin - a;
419 float u = f * dot(s, h);
420 if (u < 0.0f || u > 1.0f) {
421 return false;
422 }
423
424 vec3 q = cross(s, edge1);
425 float v = f * dot(dir, q);
426 if (v < 0.0f || u + v > 1.0f) {
427 return false;
428 }
429
430 float t = f * dot(edge2, q);
431 if (t > 0.0) {
432 out_hit.t = t;
433
434 float w = 1.0f - u - v;
435 out_hit.pos = a * w + b * u + c * v;
436 out_hit.barycentrics = vec3(w, u, v);
437 out_hit.hit = true;
438
439 return true;
440 }
441
442 return false;
443}
