222 FILE* file = fopen(filepath, "r");
223 if (!file) {
224 printf("Failed to open LUT file: %s\n", filepath);
225 return 0;
226 }
227
228 int lut_size = 0;
229 char line[256];
230
231 while (fgets(line, sizeof(line), file)) {
232 if (strncmp(line, "LUT_3D_SIZE", 11) == 0) {
233 sscanf(line, "LUT_3D_SIZE %d", &lut_size);
234 break;
235 }
236 }
237
238 if (lut_size != 33) {
239 printf("Warning: Expected LUT size 33, got %d\n", lut_size);
240 }
241
242 int total_elements = lut_size * lut_size * lut_size * 3;
243 float* lut_data = (float*)malloc(total_elements * sizeof(float));
244
245 int index = 0;
246 while (fgets(line, sizeof(line), file)) {
247 if (line[0] == '#' || line[0] == '\n' || line[0] == '\r') continue;
248
249 float r, g, b;
250 if (sscanf(line, "%f %f %f", &r, &g, &b) == 3) {
251 lut_data[index++] = r;
252 lut_data[index++] = g;
253 lut_data[index++] = b;
254 }
255 }
256 fclose(file);
257
258 GLuint lut_texture;
259 glGenTextures(1, &lut_texture);
260 glBindTexture(GL_TEXTURE_3D, lut_texture);
261
262 glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
263 glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);

265
266 glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
267 glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
268
269 glTexImage3D(GL_TEXTURE_3D, 0, GL_RGB16F, lut_size, lut_size, lut_size, 0, GL_RGB, GL_FLOAT, lut_data);
