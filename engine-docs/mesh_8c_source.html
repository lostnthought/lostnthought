460// notes on incremental hull
461// start with a point cloud of deduplicated vertices
462// mesh must be convex already (use a decomposition algo)
463// before adding new face, check length(n) < eps, if so skip (degenerate faces)
464
465// start with 4 points:
466// pick a random point
467// find furthest point from that point
468// find futhest point from that edge:
469// ab = B - A
470// ap = P - A
471// dist = length(cross(ap, ab)) / length(ab)
472// if cross(B-A,C-A) is zero, there is no triangle
473// if d < eps always, all points are colinear, this is 2D.
474
475// find furthest point from that triangle:
476 // a = reference point
477 // n = normalize(cross(B-A, C-A))
478 // dist = fabsf(dot(P-A,n))
479 // if (dist == 0) always, hull is degenerate
480// for each face:
481 // compute normal
482 // find tetrahedron vert that is not on the face (ABC would be D)
483 // d = dot(D-A, n)
484 // if d > 0, the normal must be flipped, swap two vertices
485
486// now we have a tetrahedron, compute center
487 // center = (A+B+C+D) * 0.25f
488
489// for each point:
490 // determine visibility:
491 // for each face:
492 // d = (dot(P-A, face normal))
493 // test fabsf(d) < eps
494 // if (d > 0)
495 // add to visible faces
496 // if (d < 0)
497 // not visible
498 // if (d == 0)
499 // point lies on plane
500
501 // find horizon edges:
502 // for each visible face:
503 // count edges that are shared between visible faces
504 // if count == 1, this is a horizon edge
505 // delete visible faces from master list
506 // connect P to each horizon edge and emit new faces to master list
507 // for each new face, compute normal and verify it faces correctly:
508 // if (dot(center - face_vertex, n) > 0) flip normal (swap two vertices)

514int pick_lowest_point(ConvexHull * hull)
515{
516 AXIS longest;
517 if (hull->aabb.extents.x >= hull->aabb.extents.y && hull->aabb.extents.x >= hull->aabb.extents.z) longest = X;
518 if (hull->aabb.extents.y >= hull->aabb.extents.x && hull->aabb.extents.y >= hull->aabb.extents.z) longest = Y;
519 if (hull->aabb.extents.z >= hull->aabb.extents.x && hull->aabb.extents.z >= hull->aabb.extents.y) longest = Z;
520
521 int index = 0;
522 float lowest = INFINITY;
523 vec3 * points = CAST_VEC(hull->vertices, vec3);
524 switch(longest)
525 {
526 case X:
527 {
528 for (int i = 0; i < hull->vertices.count; i++)
529 {
530 if (points[i].x < lowest) {
531 lowest = points[i].x;
532 index = i;
533 }
534 }
535 }
536 break;
537 case Y:
538 for (int i = 0; i < hull->vertices.count; i++)
539 {
540 if (points[i].y < lowest) {
541 lowest = points[i].y;
542 index = i;
543 }
544 }
545 break;
546 case Z:
547 for (int i = 0; i < hull->vertices.count; i++)
548 {
549 if (points[i].z < lowest) {
550 lowest = points[i].z;
551 index = i;
552 }
553 }
554 break;
555 }
556
557 return index;
558
