445 case JUSTIFY_LEFT:

611void process_ui_event(UICallback * callback)
612{
613
614 UIValue * val = &callback->val;
615 switch(callback->val.type)
616 {
617 case UIVAL_NONE:
618 break;
619 case UIVAL_BOOL:
620 *(bool*)callback->val.data = !(*(bool*)callback->val.data);
621 break;
622 case UIVAL_INT:
623 {
624 switch(val->behavior)
625 {
626 case UIVALBEHAVIOR_NONE:
627 *(int*)val->data = (*(int*)val->data) + val->inc_val.inc_int;
628 break;
629 case UIVALBEHAVIOR_WRAP:
630 *(int*)val->data = ((*(int*)val->data) + val->inc_val.inc_int + val->max_val.max_int) % val->max_val.max_int;
631 break;
632 case UIVALBEHAVIOR_CLAMP:
633 *(int*)val->data = MAX(MIN(((*(int*)val->data) + val->inc_val.inc_int), val->max_val.max_int), val->min_val.min_int);
634 break;
635 }
636 }
637 break;
638 case UIVAL_FLOAT:
639 {
640
641 switch(val->behavior)
642 {
643 case UIVALBEHAVIOR_NONE:
644 *(float*)val->data = ((*(float*)val->data) + val->inc_val.inc_flt);
645 break;
646 case UIVALBEHAVIOR_WRAP:
647 *(float*)val->data = fmodf(((*(float*)val->data) + val->inc_val.inc_flt + val->max_val.max_flt), val->max_val.max_flt);
648 break;
649 case UIVALBEHAVIOR_CLAMP:
650 *(float*)val->data = fmaxf(fminf(((*(float*)val->data) + val->inc_val.inc_flt), val->max_val.max_flt), val->min_val.min_flt);
651 break;
652 }
653 }
654 break;
655 case UIVAL_CALLBACK:
656 if (callback->action_fn) {
657 callback->action_fn(callback->val.data);
658 }
659 break;
660 }
661}

696void handle_ui_event(SDL_Event event, int mouse_x, int mouse_y)
697{
698
699 int menu_idx = -1;
700 int highest_ui_element_idx = get_ui_element_from_coords((vec2){mouse_x, mouse_y}, &menu_idx);
701
702 UIMenu * ms = CAST_VEC(MENUS, UIMenu);
703
704
705 UIElement * highest_el = NULL;
706 if (menu_idx != -1 && highest_ui_element_idx != -1)
707 {
708 UIElement * els = ms[menu_idx].elements;
709 highest_el = &els[highest_ui_element_idx];
710 }
711
712
713 for (int i = 0; i < MENUS.count; i++)
714 {
715 UIMenu * m = &ms[i];
716
717 UIElement * els = m->elements;
718
719 for (int j = 0; j < MAX_UI_ELEMENTS; j++)
720 {
721
722 UIElement * el = &els[j];
723
724 if (!el->visible || el->free) continue;
725
726 bool hovering = is_hovering_ui_element(el, (vec2){mouse_x, mouse_y});
727 bool processed_press = false;
728 if (highest_el)
729 {
730 if (highest_el->uID == el->uID && menu_idx == i && event.type == SDL_MOUSEBUTTONDOWN)
731 {
732 processed_press = true;
733 if (el->action_callback.one_shot)
734 {
735
736 if (el->state == UISTATE_PRESSED)
737 {
738 if (hovering)
739 {
740 process_ui_event(&el->hover_callback);
741 el->state = UISTATE_HOVERED;
742 }
743 else
744 {
745 el->state = UISTATE_NONE;
746 }
747 }
748 else
749 {
750
751 process_ui_event(&el->action_callback);
752 el->state = UISTATE_NONE;
753 }
754 }
755 else
756 {
757 process_ui_event(&el->action_callback);
758 el->state = UISTATE_PRESSED;
759 }
760 }
761
762 }
763 if (!processed_press)
764 {
765
766 if (hovering)
767 {
768 if (event.type != SDL_MOUSEBUTTONUP && el->state == UISTATE_PRESSED && !el->action_callback.one_shot) continue;
769 if (el->already_hovered) continue;
770
771 if (el->hover_callback.one_shot)
772 {
773
774 if (el->state == UISTATE_HOVERED)
775 {
776
777 el->already_hovered = true;
778 el->state = UISTATE_NONE;
779 }
780 else
781 {
782
783 process_ui_event(&el->hover_callback);
784 el->state = UISTATE_NONE;
785 }
786 }
787 else
788 {
789
790 process_ui_event(&el->hover_callback);
791 el->state = UISTATE_HOVERED;
792 }
793 }
794 else
795 {
796
797 if (event.type != SDL_MOUSEBUTTONUP && el->state == UISTATE_PRESSED && !el->action_callback.one_shot) continue;
798
799 if (el->state == UISTATE_HOVERED)
800 {
801 el->already_hovered = false;
802 }
803 el->state = UISTATE_NONE;
804
805 }
806 }
807
808 }
