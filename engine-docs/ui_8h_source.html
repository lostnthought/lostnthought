875static inline UIMenu * create_menu() {
876
877 UIMenu * menu = vector_expand(&MENUS, 1);
878
879 for (int i = 0; i < MAX_UI_ELEMENTS; i++)
880 {
881 menu->elements[i].free = true;
882 }
883 for (int i = 0; i < MAX_TEXT_BOXES; i++)
884 {
885 menu->text_boxes[i].free = true;
886 }
887 for (int i = 0; i < MAX_UI_CONTAINERS; i++)
888 {
889 menu->containers[i].free = true;
890 }
891
892 menu->uID = MENUS.count-1;
893 menu->id[0] = '\0';
894
895 return menu;
896}

905static inline bool container_is_valid(int container, int menu)
906{
907 if (menu < 0 || menu >= MENUS.count) return false;
908
909 UIMenu * ms = CAST_VEC(MENUS, UIMenu);
910 UIMenu * m = &ms[menu];
911
912 if (container >= 0 && container < MAX_UI_CONTAINERS)
913 {
914
915 UIContainer * conts = m->containers;
916
917 UIContainer * c = &conts[container];
918 if (c->uID != -1 && !c->free)
919 {
920 return true;
921 }
922
923 }
924
925 return false;
926
927}
