558static inline bool container_is_valid(int container, int menu)
559{
560 if (menu < 0 || menu >= MENUS.count) return false;
561
562 UIMenu * ms = CAST_VEC(MENUS, UIMenu);
563 UIMenu * m = &ms[menu];
564
565 if (container >= 0 && container < MAX_UI_CONTAINERS)
566 {
567
568 UIContainer * conts = m->containers;
569
570 UIContainer * c = &conts[container];
571 if (c->uID != -1 && !c->free)
572 {
573 return true;
574 }
575
576 }
577
578 return false;
