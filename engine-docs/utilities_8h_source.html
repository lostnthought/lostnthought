329static inline mat3 quat_to_mat3(const vec4 q) {
330 float w = q.x;
331 float x = q.y;
332 float y = q.z;
333 float z = q.w;
334
335 float x2 = x * x;
336 float y2 = y * y;
337 float z2 = z * z;
338 float xy = x * y;
339 float xz = x * z;
340 float yz = y * z;
341 float wx = w * x;
342 float wy = w * y;
343 float wz = w * z;
344
345 mat3 mat;
346 mat.a0 = 1.0f - 2.0f * (y2 + z2);
347 mat.a1 = 2.0f * (xy - wz);
348 mat.a2 = 2.0f * (xz + wy);
349
350 mat.b0 = 2.0f * (xy + wz);
351 mat.b1 = 1.0f - 2.0f * (x2 + z2);
352 mat.b2 = 2.0f * (yz - wx);
353
354 mat.c0 = 2.0f * (xz - wy);
355 mat.c1 = 2.0f * (yz + wx);
356 mat.c2 = 1.0f - 2.0f * (x2 + y2);
357 return mat;
358}

459static inline mat4 mat4_from_quat(vec4 q) {
460 float x = q.x, y = q.y, z = q.z, w = q.w;
461 float xx = x * x, yy = y * y, zz = z * z;
462 float xy = x * y, xz = x * z, yz = y * z;
463 float wx = w * x, wy = w * y, wz = w * z;
464
465 mat4 m = mat4_identity();
466
467 m.a0 = 1.0f - 2.0f * (yy + zz);
468 m.a1 = 2.0f * (xy + wz);
469 m.a2 = 2.0f * (xz - wy);
470
471 m.b0 = 2.0f * (xy - wz);
472 m.b1 = 1.0f - 2.0f * (xx + zz);
473 m.b2 = 2.0f * (yz + wx);
474
475 m.c0 = 2.0f * (xz + wy);
476 m.c1 = 2.0f * (yz - wx);
477 m.c2 = 1.0f - 2.0f * (xx + yy);
478
479 return m;

559static inline mat3 mat3_mul(mat3 M, mat3 N) {
560 mat3 R;
561
562 R.a0 = M.a0 * N.a0 + M.a1 * N.b0 + M.a2 * N.c0;
563 R.a1 = M.a0 * N.a1 + M.a1 * N.b1 + M.a2 * N.c1;

576static inline mat3 mat3_inverse_transpose(mat3 m) {
577 float det =
578 m.a0*(m.b1*m.c2 - m.b2*m.c1) -
579 m.a1*(m.b0*m.c2 - m.b2*m.c0) +
580 m.a2*(m.b0*m.c1 - m.b1*m.c0);
581
582 float invDet = (det != 0.0f) ? 1.0f/det : 1.0f;
583
