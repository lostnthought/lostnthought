116float noise(vec3 p){
117 vec3 a = floor(p);
118 vec3 d = p - a;
119 d = d * d * (3.0 - 2.0 * d);
120
121 vec4 b = a.xxyy + vec4(0.0, 1.0, 0.0, 1.0);
122 vec4 k1 = perm(b.xyxy);
123 vec4 k2 = perm(k1.xyxy + b.zzww);
124
125 vec4 c = k2 + a.zzzz;
126 vec4 k3 = perm(c);
127 vec4 k4 = perm(c + 1.0);
128
129 vec4 o1 = fract(k3 * (1.0 / 41.0));
130 vec4 o2 = fract(k4 * (1.0 / 41.0));
131
132 vec4 o3 = o2 * d.z + o1 * (1.0 - d.z);
133 vec2 o4 = o3.yw * d.x + o3.xz * (1.0 - d.x);
134
135 return o4.y * d.y + o4.x * (1.0 - d.y);
136}
