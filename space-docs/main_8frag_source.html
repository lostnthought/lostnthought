101float noise(vec3 p){
102 vec3 a = floor(p);
103 vec3 d = p - a;
104 d = d * d * (3.0 - 2.0 * d);
105
106 vec4 b = a.xxyy + vec4(0.0, 1.0, 0.0, 1.0);
107 vec4 k1 = perm(b.xyxy);
108 vec4 k2 = perm(k1.xyxy + b.zzww);
109
110 vec4 c = k2 + a.zzzz;
111 vec4 k3 = perm(c);
112 vec4 k4 = perm(c + 1.0);
113
114 vec4 o1 = fract(k3 * (1.0 / 41.0));
115 vec4 o2 = fract(k4 * (1.0 / 41.0));
116
117 vec4 o3 = o2 * d.z + o1 * (1.0 - d.z);
118 vec2 o4 = o3.yw * d.x + o3.xz * (1.0 - d.x);

120 return o4.y * d.y + o4.x * (1.0 - d.y);
121}
