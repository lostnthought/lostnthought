115float noise(vec3 p){
116 vec3 a = floor(p);
117 vec3 d = p - a;
118 d = d * d * (3.0 - 2.0 * d);
119
120 vec4 b = a.xxyy + vec4(0.0, 1.0, 0.0, 1.0);
121 vec4 k1 = perm(b.xyxy);
122 vec4 k2 = perm(k1.xyxy + b.zzww);
123
124 vec4 c = k2 + a.zzzz;
125 vec4 k3 = perm(c);
126 vec4 k4 = perm(c + 1.0);
127
128 vec4 o1 = fract(k3 * (1.0 / 41.0));
129 vec4 o2 = fract(k4 * (1.0 / 41.0));
130
131 vec4 o3 = o2 * d.z + o1 * (1.0 - d.z);
132 vec2 o4 = o3.yw * d.x + o3.xz * (1.0 - d.x);
133
134 return o4.y * d.y + o4.x * (1.0 - d.y);
135}
