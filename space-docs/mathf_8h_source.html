112
113 vec3 a = { in.a0, in.a1, in.a2 }; // X basis
114 vec3 b = { in.b0, in.b1, in.b2 }; // Y basis
115 vec3 c = { in.c0, in.c1, in.c2 }; // Z basis
116 vec3 d = { in.d0, in.d1, in.d2 }; // Translation
117
118 vec3 cross_bc = vec3_cross(b, c);

120 float invDet = 1.0f / det;
121 vec3 ia = vec3_cross(b, c);
122 vec3 ib = vec3_cross(c, a);
123 vec3 ic = vec3_cross(a, b);
124
125 ia.x *= invDet; ia.y *= invDet; ia.z *= invDet;
126 ib.x *= invDet; ib.y *= invDet; ib.z *= invDet;
127 ic.x *= invDet; ic.y *= invDet; ic.z *= invDet;
128
129 vec3 id = {
130 -vec3_dot(ia, d),
131 -vec3_dot(ib, d),
132 -vec3_dot(ic, d)
133 };

135 mat4 out;
