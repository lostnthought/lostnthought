7char *read_file(const char *filename) {
8 FILE *f = fopen(filename, "rb");
9 if (!f) return NULL;
10 fseek(f, 0, SEEK_END);
11 long size = ftell(f);
12 rewind(f);
13 char *buffer = (char *)malloc(size + 1);
14 if (!buffer) return NULL;
15 fread(buffer, 1, size, f);
16 buffer[size] = '\0';
17 fclose(f);
18 return buffer;
19}
20GLuint create_compute_shader(const char *filename) {
21 char *src = read_file(filename);
22 if (!src) {
23 fprintf(stderr, "Failed to read shader file: %s\n", filename);
24 return 0;
25 }
26
27 GLuint shader = glCreateShader(GL_COMPUTE_SHADER);
28 glShaderSource(shader, 1, (const GLchar * const *)&src, NULL);
29 glCompileShader(shader);
30 free(src); // don't forget to free it after compile
31
32 GLint status;
33 glGetShaderiv(shader, GL_COMPILE_STATUS, &status);
34 if (!status) {
35 char log[512];
36 glGetShaderInfoLog(shader, 512, NULL, log);
37 fprintf(stderr, "Compute Shader Error %s: %s\n", filename, log);
38 exit(0);
39 return 0;
40 }
41
42 GLuint program = glCreateProgram();
43 glAttachShader(program, shader);
44 glLinkProgram(program);
45
46 GLint status2;
47 glGetProgramiv(program, GL_LINK_STATUS, &status2);
48 if (!status2) {
49 char log[512];
50 glGetProgramInfoLog(program, 512, NULL, log);
51 fprintf(stderr, "Compute Shader LINKER Error %s: %s\n", filename, log);
52 exit(0);
53 return 0;
54 }
55
56 glDeleteShader(shader);
57 return program;
58}

61GLuint compile_shader(GLenum type, const char *filename) {
62 const char *source = read_file(filename);
63 if (!source) return 0;
64
65 GLuint shader = glCreateShader(type);
66 glShaderSource(shader, 1, &source, NULL);
67 glCompileShader(shader);
68 free((void *)source);
69
70 GLint success;
71 glGetShaderiv(shader, GL_COMPILE_STATUS, &success);
72 char log[512];
73 if (!success) {
74
75 glGetShaderInfoLog(shader, 512, NULL, log);
76 printf("\\nShader compile error (%s):\\n%s\\n", filename, log);
77 exit(0);
78 return 0;
79 } else {
80 glGetShaderInfoLog(shader, 512, NULL, log);
81 if (strlen(log) > 0) {
82 printf("Shader compile warning (%s):\\n%s\\n", filename, log);
83 }
84 }
85
86 return shader;
87}

91GLuint create_shader_program(const char *vertexPath, const char *fragmentPath) {
92 GLuint vertexShader = compile_shader(GL_VERTEX_SHADER, vertexPath);
93 GLuint fragmentShader = compile_shader(GL_FRAGMENT_SHADER, fragmentPath);
94 if (!vertexShader || !fragmentShader) return 0;
95
96 GLuint program = glCreateProgram();
97 glAttachShader(program, vertexShader);
98 glAttachShader(program, fragmentShader);
99 glLinkProgram(program);
100
101 GLint success;
102 glGetProgramiv(program, GL_LINK_STATUS, &success);
103 char log[512];
104 if (!success) {
105
106 glGetProgramInfoLog(program, 512, NULL, log);
107 printf("Shader link error:\\n%s\\n", log);
108 exit(0);
109 return 0;
110 } else {
111 glGetProgramInfoLog(program, 512, NULL, log);
112 if (strlen(log) > 0) {
113 printf("Shader link warning:\\n%s\\n", log);
114 }
115}
116
117 glDeleteShader(vertexShader);
118 glDeleteShader(fragmentShader);
119
120 return program;
121}
