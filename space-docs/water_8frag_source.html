145float noise(vec3 p){
146 vec3 a = floor(p);
147 vec3 d = p - a;
148 d = d * d * (3.0 - 2.0 * d);
149
150 vec4 b = a.xxyy + vec4(0.0, 1.0, 0.0, 1.0);
151 vec4 k1 = perm(b.xyxy);
152 vec4 k2 = perm(k1.xyxy + b.zzww);
153
154 vec4 c = k2 + a.zzzz;
155 vec4 k3 = perm(c);
156 vec4 k4 = perm(c + 1.0);

158 vec4 o1 = fract(k3 * (1.0 / 41.0));
159 vec4 o2 = fract(k4 * (1.0 / 41.0));

161 vec4 o3 = o2 * d.z + o1 * (1.0 - d.z);

187{
188 vec3 proj_coords = pos.xyz / pos.w;

195 float shadow = 0.0f;

197 for (int y = -1; y <= 1; y++)

199
